    async function joinBattle(battleId,battleData){
    if(!myUid)return;
    const total=battleData.totalCost||0;
    if(getCredits()<total){alert('Not enough credits! Need '+total.toLocaleString());return;}

    const snap=await rtdb.ref('battles/'+battleId).once('value');
    const b=snap.val();
    if(!b||b.status!=='waiting'){alert('Battle no longer available.');return;}
    if(Object.keys(b.players||{}).length>=2){alert('Battle is full.');return;}

    // Always fetch the latest nickname from the database
    const userSnap=await rtdb.ref('users/'+myUid).once('value');
    const userData=userSnap.val()||{};
    const nickname=userData.username||currentUser.displayName||currentUser.email||'Player';
    const avatar=userData.avatar||currentUser.photoURL||'';

    await adjCredits(myUid,-total);

    await rtdb.ref('battles/'+battleId+'/players/'+myUid).set({
      uid:myUid,name:nickname,avatar:avatar,total:0,items:{},left:false
    });
    await rtdb.ref('battles/'+battleId+'/status').set('playing');

    openBattleRoom(battleId);
  }

     function markLeft(){
    if(activeBattle&&myUid){
      rtdb.ref('battles/'+activeBattle+'/playersLeft/'+myUid).set(true);

      // If game is still playing, mark forfeit
      if(battleState&&battleState.status==='playing'&&!battleState.winner){
        const pids=Object.keys(battleState.players||{});
        const opponentId=pids.find(id=>id!==myUid);
        rtdb.ref('battles/'+activeBattle+'/winner').set(opponentId||'none');
        rtdb.ref('battles/'+activeBattle+'/status').set('finished');
        rtdb.ref('battles/'+activeBattle+'/forfeitedBy').set(myUid);
        if(opponentId){
          const bet=battleState.bet||0;
          const allSkinsValue=pids.reduce((sum,pid)=>sum+((battleState.players[pid]||{}).total||0),0);
          adjCredits(opponentId, bet*2 + allSkinsValue);
        }
        trackForfeit(myUid);
      }
    }
  }

    function closeBattleRoom(){
    if(battleRef&&battleUnsub){
      battleRef.off('value',battleUnsub);
    }

       // If battle is still playing and I leave = forfeit
    if(activeBattle&&myUid&&battleState&&battleState.status==='playing'&&!battleState.winner){
      const pids=Object.keys(battleState.players||{});
      const opponentId=pids.find(id=>id!==myUid);

      // Mark me as forfeited
      rtdb.ref('battles/'+activeBattle+'/winner').set(opponentId||'none');
      rtdb.ref('battles/'+activeBattle+'/status').set('finished');
      rtdb.ref('battles/'+activeBattle+'/forfeitedBy').set(myUid);

      // Opponent gets both bets + all skin value from both players
      if(opponentId){
        const bet=battleState.bet||0;
        const allSkinsValue=pids.reduce((sum,pid)=>sum+((battleState.players[pid]||{}).total||0),0);
        adjCredits(opponentId, bet*2 + allSkinsValue);
      }

      // Track daily forfeits
      trackForfeit(myUid);
    }

    if(activeBattle&&myUid){
      rtdb.ref('battles/'+activeBattle+'/playersLeft/'+myUid).set(true);
      rtdb.ref('battles/'+activeBattle+'/playersLeft').once('value').then(snap=>{
        const left=snap.val()||{};
        const allLeft=Object.values(left).every(v=>v===true)&&Object.keys(left).length>=2;
        if(allLeft) rtdb.ref('battles/'+activeBattle).remove();
      });
    }

    window.removeEventListener('beforeunload',markLeft);
    activeBattle=null;battleRef=null;battleUnsub=null;battleState=null;spinning=false;roundsAnimated=new Set();
    document.getElementById('battleOverlay').classList.remove('open');
    document.getElementById('mainLobby').style.display='';
  }

    async function trackForfeit(uid){
    const today=new Date().toISOString().slice(0,10); // "2026-03-30"
    const ref=rtdb.ref('users/'+uid+'/forfeits/'+today);
    const snap=await ref.once('value');
    const count=(snap.val()||0)+1;
    await ref.set(count);

    // 3+ forfeits in a day → flag to support
    if(count>=3){
      const userName=myName||uid;
      await rtdb.ref('reports').push().set({
        targetType:'user',
        targetId:uid,
        targetAuthorUid:uid,
        reporterUid:'system',
        reason:`Auto-flag: ${userName} forfeited ${count} case battles today (${today}). Possible abuse.`,
        timestamp:Date.now(),
        resolved:false,
        autoGenerated:true
      });
    }
  }

  function pickItem(items){
    const tw=items.reduce((s,i)=>s+i.weight,0);
    let r=Math.random()*tw;
    for(const it of items){r-=it.weight;if(r<=0)return it;}
    return items[items.length-1];
  }

  async function generateRound(data,round){
    if(spinning)return;
    const caseId=data.cases[round];
    const cd=CASE_MAP[caseId];
    if(!cd)return;

    const pids=Object.keys(data.players);
    const results={};
    pids.forEach(pid=>{
      const won=pickItem(cd.items);
      results[pid]={name:won.name,img:won.img,value:won.value,rarity:won.rarity};
    });

    await rtdb.ref('battles/'+activeBattle+'/rounds/'+round).set(results);
    // Don't advance currentRound here — it stays the same, the on('value') listener
    // will re-fire and both clients will see the round data and start animating
  }

  async function playRound(round){
    if(spinning)return;
    spinning=true;
    roundsAnimated.add(round);

    // Always read fresh state from Firebase
    const snap=await rtdb.ref('battles/'+activeBattle).once('value');
    const data=snap.val();
    if(!data||data.status!=='playing'){spinning=false;return;}

    const caseId=data.cases[round];
    const cd=CASE_MAP[caseId];
    if(!cd){spinning=false;return;}

    const roundData=(data.rounds||{})[round];
    if(!roundData){spinning=false;return;}

    const pids=Object.keys(data.players);
    const p1id=data.creator;
    const p2id=pids.find(id=>id!==data.creator);

    const p1Win=roundData[p1id];
    const p2Win=roundData[p2id];
    if(!p1Win||!p2Win){spinning=false;return;}

    const p1Spinner=document.getElementById('p1Spinner');
    const p2Spinner=document.getElementById('p2Spinner');
    const p1Track=document.getElementById('p1Track');
    const p2Track=document.getElementById('p2Track');

    if(!p1Spinner||!p2Spinner){spinning=false;return;}

    p1Spinner.style.display='block';
    p2Spinner.style.display='block';

    buildTrack(p1Track,cd,p1Win);
    buildTrack(p2Track,cd,p2Win);

    await Promise.all([
      animateTrack(p1Spinner,p1Track),
      animateTrack(p2Spinner,p2Track)
    ]);

    await delay(600);

    // Show won items in the list for both clients
    const p1Items=document.getElementById('p1Items');
    const p2Items=document.getElementById('p2Items');
    if(p1Items&&p1Win){
      const rar=RARITIES[p1Win.rarity]||RARITIES.common;
      p1Items.insertAdjacentHTML('beforeend',`<div class="pp-item"><span class="pp-item-dot" style="background:${rar.color}"></span><img src="${p1Win.img}" alt="" onerror="this.style.display='none'"/><span class="pp-item-name">${p1Win.name}</span><span class="pp-item-val">🪙 ${(p1Win.value||0).toLocaleString()}</span></div>`);
    }
    if(p2Items&&p2Win){
      const rar=RARITIES[p2Win.rarity]||RARITIES.common;
      p2Items.insertAdjacentHTML('beforeend',`<div class="pp-item"><span class="pp-item-dot" style="background:${rar.color}"></span><img src="${p2Win.img}" alt="" onerror="this.style.display='none'"/><span class="pp-item-name">${p2Win.name}</span><span class="pp-item-val">🪙 ${(p2Win.value||0).toLocaleString()}</span></div>`);
    }

    // Update totals visually
    const p1Total=(data.players[p1id]?.total||0)+p1Win.value;
    const p2Total=(data.players[p2id]?.total||0)+(p2Win?p2Win.value:0);
    const p1TotalEl=document.querySelector('#p1Panel .pp-total');
    const p2TotalEl=document.querySelector('#p2Panel .pp-total');
    if(p1TotalEl) p1TotalEl.textContent='🪙 '+p1Total.toLocaleString();
    if(p2TotalEl) p2TotalEl.textContent='🪙 '+p2Total.toLocaleString();

    // Hide spinners
    p1Spinner.style.display='none';
    p2Spinner.style.display='none';

       // Only the creator advances the game state in the database
    if(myUid===data.creator){
      // Write player items and totals
      for(const pid of pids){
        const item=roundData[pid];
        const curTotal=(data.players[pid]?.total||0);
        await rtdb.ref('battles/'+activeBattle+'/players/'+pid+'/items/'+round).set(item);
        await rtdb.ref('battles/'+activeBattle+'/players/'+pid+'/total').set(curTotal+item.value);
      }

      const nextRound=round+1;

      if(nextRound>=data.cases.length){
        // Last round — finish the battle, then set currentRound
        await rtdb.ref('battles/'+activeBattle+'/currentRound').set(nextRound);
        await finishBattle(data,nextRound);
        // Small delay to let Firebase propagate the winner/status writes
        await delay(300);
      }else{
        // Generate next round's results FIRST, then advance currentRound
        const nextCaseId=data.cases[nextRound];
        const nextCd=CASE_MAP[nextCaseId];
        if(nextCd){
          const nextResults={};
          pids.forEach(pid=>{
            const won=pickItem(nextCd.items);
            nextResults[pid]={name:won.name,img:won.img,value:won.value,rarity:won.rarity};
          });
          await rtdb.ref('battles/'+activeBattle+'/rounds/'+nextRound).set(nextResults);
        }
        await rtdb.ref('battles/'+activeBattle+'/currentRound').set(nextRound);
      }
    }

    spinning=false;

    // Both clients: poll until we get a definitive next state
    // This handles the case where on('value') fired while spinning was true
    const pollForNext=async(retries)=>{
      for(let i=0;i<retries;i++){
        await delay(500);
        if(!activeBattle)return;
        const freshSnap=await rtdb.ref('battles/'+activeBattle).once('value');
        const freshData=freshSnap.val();
        if(!freshData)return;

        // Game finished — re-render to show results
        if(freshData.status==='finished'||freshData.winner){
          renderBattle(freshData);
          return;
        }

        // Next round is ready — play it
        if(freshData.status==='playing'){
          const nextR=freshData.currentRound||0;
          const nextRoundData=(freshData.rounds||{})[nextR];
          if(nextR<freshData.cases.length&&nextRoundData&&!roundsAnimated.has(nextR)){
            await delay(300);
            playRound(nextR);
            return;
          }
          // If all rounds are done but status isn't finished yet, the creator
          // is still writing — keep polling
          if(nextR>=freshData.cases.length){
            continue; // retry, finishBattle hasn't completed yet
          }
        }
      }
      // Final fallback: force a re-read and render whatever we have
      if(!activeBattle)return;
      const finalSnap=await rtdb.ref('battles/'+activeBattle).once('value');
      const finalData=finalSnap.val();
      if(finalData) renderBattle(finalData);
    };

    await pollForNext(10);
  }

  function buildTrack(trackEl,caseData,winItem){
    trackEl.innerHTML='';
    trackEl.style.transition='none';
    trackEl.style.transform='translateX(0)';
    const ITEM_W=window.innerWidth<=768?75:90;
    const TOTAL=55;
    const WIN_POS=45;

    for(let i=0;i<TOTAL;i++){
      const item=i===WIN_POS?winItem:caseData.items[Math.floor(Math.random()*caseData.items.length)];
      const r=RARITIES[item.rarity]||RARITIES.common;
      const el=document.createElement('div');
      el.className='b-track-item';
      el.style.width=ITEM_W+'px';
      el.innerHTML=`<img src="${item.img}" alt="" onerror="this.style.display='none'" style="filter:drop-shadow(0 0 5px ${r.color})"/><div class="bti-name">${item.name}</div><div class="bti-val">🪙 ${item.value}</div><div style="position:absolute;bottom:0;left:0;right:0;height:3px;background:${r.color};opacity:.5"></div>`;
      trackEl.appendChild(el);
    }
  }

  async function animateTrack(spinnerEl,trackEl){
    const ITEM_W=window.innerWidth<=768?75:90;
    const WIN_POS=45;
    void trackEl.offsetHeight;
    const areaW=spinnerEl.offsetWidth;
    const targetX=(WIN_POS*ITEM_W)+(ITEM_W/2)-(areaW/2);
    const offset=(Math.random()-0.5)*(ITEM_W*0.35);
    trackEl.style.transition='transform 4.5s cubic-bezier(0.12, 0.87, 0.22, 1)';
    trackEl.style.transform=`translateX(-${targetX+offset}px)`;
    await delay(4800);
  }

  function delay(ms){return new Promise(r=>setTimeout(r,ms));}

  /* ═══════════════════════════════════════════
     FINISH BATTLE
     ═══════════════════════════════════════════ */
   async function finishBattle(data,totalRoundsDone){
    // Re-read latest
    const snap=await rtdb.ref('battles/'+activeBattle+'/players').once('value');
    const ps=snap.val()||{};
    const pids=Object.keys(ps);
    const totals={};
    pids.forEach(pid=>totals[pid]=ps[pid].total||0);

    // Total skin value from ALL players combined
    const allSkinsValue=pids.reduce((sum,pid)=>sum+(ps[pid].total||0),0);

    const sorted=pids.sort((a,b)=>totals[b]-totals[a]);
    const bet=data.bet||0;
    let winner;

    if(pids.length===2&&totals[sorted[0]]===totals[sorted[1]]){
      winner='draw';
      // Return bets + each player gets their own skin value back
      for(const pid of pids) await adjCredits(pid, bet + (ps[pid].total||0));
    }else{
      winner=sorted[0];
      // Winner gets: both side bets + ALL skin value from both players
      await adjCredits(winner, bet*2 + allSkinsValue);
    }

    await rtdb.ref('battles/'+activeBattle+'/winner').set(winner);
    await rtdb.ref('battles/'+activeBattle+'/status').set('finished');
  }

  /* ═══════════════════════════════════════════
     CLEANUP — check on page load for leftover battles
     ═══════════════════════════════════════════ */
  // Also clean up battles where both left
  setInterval(async()=>{
    if(!myUid)return;
    const snap=await rtdb.ref('battles').once('value');
    const all=snap.val()||{};
    for(const[id,b]of Object.entries(all)){
      // Delete if both players left
      const left=b.playersLeft||{};
      const playerCount=Object.keys(b.players||{}).length;
      const leftCount=Object.values(left).filter(v=>v===true).length;
      if(playerCount>=2&&leftCount>=2){
