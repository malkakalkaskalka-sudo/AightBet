     ═══════════════════════════════════════════ */
  cashoutBtn.addEventListener('click',async function(){
    if(!playing||currentLane<0) return;
    crossBtn.disabled=true;
    cashoutBtn.disabled=true;

    var m=getMult();
    var w=Math.round(bet*m);
    await adjustCredits(w);

    totalGames++;
    totalSurvived++;
    if(m>bestMult) bestMult=m;
    if(w>biggestWin) biggestWin=w;

    resultEl.innerHTML='<span class="result-cash">CASHED OUT at x'+m+'! +'+w.toLocaleString()+' credits</span>';
    statWin.textContent=w.toLocaleString();
    endGame();
  });

  /* ═══════════════════════════════════════════
     END GAME
     ═══════════════════════════════════════════ */
  function endGame(){
    playing=false;
    crossBtn.disabled=true;
    cashoutBtn.disabled=true;
    startBtn.disabled=false;
    betArea.style.opacity='1';
    betArea.querySelectorAll('button').forEach(function(b){b.disabled=false;});
    betInput.disabled=false;
    statGames.textContent=totalGames;
    statSurvived.textContent=totalSurvived;
    statBestMult.textContent='x'+bestMult;
    statBiggest.textContent=biggestWin.toLocaleString();
  }

  /* ═══════════════════════════════════════════
     PARTICLES
     ═══════════════════════════════════════════ */
  (function(){
    var canvas=document.getElementById('particles');
    var ctx=canvas.getContext('2d');
    var W=0;
    var H=0;
    var pts=[];
    var ms={x:-9999,y:-9999};
    var COUNT=Math.min(60,Math.round(window.innerWidth*0.04));
    var COLORS=['rgba(139,92,246,','rgba(6,182,212,','rgba(236,72,153,'];

    function resize(){
      var d=Math.min(devicePixelRatio||1,2);
      W=innerWidth;
      H=innerHeight;
      canvas.width=W*d;
      canvas.height=H*d;
      canvas.style.width=W+'px';
      canvas.style.height=H+'px';
      ctx.setTransform(d,0,0,d,0,0);
    }

    function mk(){
      return {
        x:Math.random()*W,
        y:Math.random()*H,
        r:Math.random()*1.8+0.4,
        vx:(Math.random()-0.5)*0.25,
        vy:(Math.random()-0.5)*0.25,
        a:Math.random()*0.4+0.15,
        c:COLORS[Math.floor(Math.random()*COLORS.length)]
      };
    }

    function init(){
      resize();
      pts=[];
      for(var i=0;i<COUNT;i++) pts.push(mk());
    }

    function draw(){
      ctx.clearRect(0,0,W,H);
      for(var i=0;i<pts.length;i++){
        var a=pts[i];
        a.x+=a.vx;
        a.y+=a.vy;
        var dx=a.x-ms.x;
        var dy=a.y-ms.y;
        var d=dx*dx+dy*dy;
        if(d<10000){
          var dist=Math.sqrt(d);
          a.vx+=dx*0.015/dist;
          a.vy+=dy*0.015/dist;
        }
        a.vx*=0.99;
        a.vy*=0.99;
        if(a.x<-10) a.x=W+10;
        if(a.x>W+10) a.x=-10;
        if(a.y<-10) a.y=H+10;
        if(a.y>H+10) a.y=-10;
        ctx.beginPath();
        ctx.arc(a.x,a.y,a.r,0,Math.PI*2);
        ctx.fillStyle=a.c+a.a+')';
        ctx.fill();
        for(var j=i+1;j<pts.length;j++){
          var b=pts[j];
          var ld=Math.pow(a.x-b.x,2)+Math.pow(a.y-b.y,2);
          if(ld<16000){
            ctx.beginPath();
            ctx.moveTo(a.x,a.y);
            ctx.lineTo(b.x,b.y);
            ctx.strokeStyle=a.c+(0.06*(1-ld/16000))+')';
            ctx.lineWidth=0.4;
            ctx.stroke();
          }
        }
      }
      requestAnimationFrame(draw);
    }

    addEventListener('resize',resize);
    addEventListener('mousemove',function(e){ms.x=e.clientX;ms.y=e.clientY;});
    addEventListener('touchmove',function(e){ms.x=e.touches[0].clientX;ms.y=e.touches[0].clientY;},{passive:true});
    init();
    draw();
  })();
  </script>
